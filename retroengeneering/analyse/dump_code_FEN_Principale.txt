nIDmachine est un entier = TABLE_Machines.COL_IDMachine

bModif est un booleen = ouvre(FEN_Machine, nIDmachine)

si bModif = vrai alors
	
	//Met à jour la table au cas où les informations de la machine aient été modifiées 
	//Si on avait un ID de lieu mémorisé : on affiche les machines de ce lieu sinon on affiche toutes les machines
	si gnIDLieu = 0 ALORS

	sinon
		AfficheMachinesLieu(gnIDLieu)
	FIN
	
	//Remet le bandeau de la table sur la machine qui a été visualisée - modifiée
	nIndice est un entier = TableCherche(COL_idMachine, nIDmachine)
	TableSelectPlus(TABLE_Machines,nIndice)
	
	//Ré-affichage du plan : si on a modifié le lieu d'une machine ...

	HLitRecherchePremier(lieu, IDLieu, machine.IDLieu)
	EntoureLieu(LIEU.Rectangle)
	
FIN

--- IMG_PlanLocaux (IMAGE) event type=18 ---


//On parcourt chaque lieu de la base de données
pour TOUT LIEU 
	
	//Si le point cliqué sur le plan correspond à un lieu donné
	si bInclusDans(lieu.Rectangle, SourisPosX(), SourisPosY()) ALORS
		
		gnIDLieu = lieu.IDLieu
		
		AffichePlan(IMG_PlanLocaux)
		
		//Entoure en rouge sur le plan le lieu sur lequel on a cliqué
		EntoureLieu(LIEU.Rectangle)		
		
		//On affiche dans la table la liste des machines de ce lieu
		AfficheMachinesLieu(lieu.IDLieu)
		
		lib_filtre = ChaîneConstruit("Machines dans la salle %1", lieu.Emplacement)
	fin

FIN

--- PROCEDURE EntoureLieu ---
// Résumé : Entoure d'un cadre rouge sur le plan le lieu correspondant au rectangle 
// Syntaxe :
//EntoureLieu (<Rectangle>)
//
// Paramètres :
//	Rectangle : coordonnées du lieu
// Valeur de retour :
// 	Aucune


Procédure EntoureLieu(Rectangle)

nLX, nLY, nRX, nRY sont des entiers
MonImagepourEntourer est une Image

dDébutDessin(MonImagepourEntourer)
MonImagepourEntourer = IMG_PlanLocaux

// Coin haut gauche du rectangle du lieu
nLX = ExtraitChaîne(Rectangle,1,",") 
nLY = ExtraitChaîne(Rectangle,2,",")

// Coin bas droit du rectangle du lieu
nRX = ExtraitChaîne(Rectangle,3,",") 
nRY = ExtraitChaîne(Rectangle,4,",")

dStylo(RougeClair, 0, 3)
dFond(Transparent)

//On dessine le lieu
dRectangle(nLX-2,nLY-2,nRX+2,nRY+2)

dFusionne(IMG_PlanLocaux, MonImagepourEntourer,50)

--- PROCEDURE AfficheMachinesLieu ---
// Résumé : Affiche dans la table toutes les machines d'un lieu donné
// Syntaxe :
//AfficheMachinesLieu (<nIDLieu>)
//
// Paramètres :
// 	nIDLieu : Identifiant du lieu pour lequel il faut rechercher les machines
// Valeur de retour :
// 	Aucune
//

Procédure AfficheMachinesLieu(nIDLieu)

//On change le contenu de la liste des machines pour y mettre les machines du lieu sur lequel l'utilisateur a cliqué
TableSupprimeTout(TABLE_Machines)

POUR TOUTE MACHINE AVEC IDLieu = nIDLieu
	
	//On affiche l'utilisateur de la machine s'il y en a un, avec un pictogramme "machine utilisée"
	//sinon on utilise un pictogramme "machine disponible"
	SI MACHINE.InitialesUtilisateur = "" ALORS
		TableAjouteLigne(TABLE_Machines, IMG_MachineDispo, MACHINE.NomMachine, "" ,MACHINE.IDMachine)
			SINON
		TableAjouteLigne(TABLE_Machines, IMG_MachineUtilisée, MACHINE.NomMachine, MACHINE.InitialesUtilisateur ,MACHINE.IDMachine)
	FIN
	
FIN

--- PROCEDURE AfficheToutesMachines ---
// Résumé : Affiche toutes les machines de tous les lieux
// Syntaxe :
// AfficheToutesMachines ()
//
// Paramètres :
//	Aucun
// Valeur de retour :
// 	Aucune

Procédure AfficheToutesMachines()


//Affiche toutes les machines de tous les lieux
TableSupprimeTout(TABLE_Machines)

POUR TOUTE MACHINE
		
	//On affiche l'utilisateur de la machine s'il y en a un, avec un pictogramme "machine utilisée"
	//sinon on utilise un pictogramme "machine disponible"
	SI MACHINE.InitialesUtilisateur = "" ALORS
		TableAjouteLigne(TABLE_Machines, IMG_MachineDispo, MACHINE.NomMachine, "" , MACHINE.IDMachine)
	SINON
		TableAjouteLigne(TABLE_Machines, IMG_MachineUtilisée, MACHINE.NomMachine, MACHINE.InitialesUtilisateur ,MACHINE.IDMachine)
	FIN
	
FIN

gnIDLieu = 0
